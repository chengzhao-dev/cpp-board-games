#===------------------------------------------------------------------------===#
# CMakeLists.txt - 棋盘状态工程的构建定义
#
# 约定产物目录，并按规则库、CLI 与测试三个组件组织构建。
#===------------------------------------------------------------------------===#

#===------------------------------------------------------------------------===#
# 工程声明
#===------------------------------------------------------------------------===#

# CMake 版本约束：保证本目录使用的命令可用。
cmake_minimum_required(VERSION 4.2)

# 项目名，只启用 C++ 语言。
project(tic_tac_toe_02_board_and_state LANGUAGES CXX)

#===------------------------------------------------------------------------===#
# 产物目录
#===------------------------------------------------------------------------===#

# 可执行文件归入 build/bin，动态库与静态库归入 build/lib。
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

#===------------------------------------------------------------------------===#
# 测试
#===------------------------------------------------------------------------===#

# 开启 CTest：子目录注册的测试才能被 ctest 发现和运行。
enable_testing()

#===------------------------------------------------------------------------===#
# 组件
#===------------------------------------------------------------------------===#

# 三个组件目录各自带 CMakeLists.txt，按依赖顺序加入构建。
add_subdirectory(tic_tac_toe)
add_subdirectory(apps)
add_subdirectory(tests)
