#===------------------------------------------------------------------------===#

#===------------------------------------------------------------------------===#
# 工程声明
#===------------------------------------------------------------------------===#

# CMake 版本约束：保证本目录使用的命令可用。
cmake_minimum_required(VERSION 4.2)

# 项目名，只启用 C++ 语言。
project(tic_tac_toe_01_toolchain_probe LANGUAGES CXX)

#===------------------------------------------------------------------------===#
# 可执行目标
#===------------------------------------------------------------------------===#

# 可执行文件统一归入 build/bin。
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

# 源文件显式列出，新增文件在这里追加。
add_executable(tic_tac_toe_cli
  src/main.cpp
)

# 这个目标使用 C++20。
target_compile_features(tic_tac_toe_cli PRIVATE cxx_std_20)

#===------------------------------------------------------------------------===#
# 测试
#===------------------------------------------------------------------------===#

# 开启 CTest：此后本目录注册的测试才能被 ctest 发现和运行。
enable_testing()

# 注册冒烟测试：返回 0 且输出含 "Hello, World!" 才通过。
add_test(
  NAME tic_tac_toe_cli_smoke
  COMMAND tic_tac_toe_cli
)
set_tests_properties(tic_tac_toe_cli_smoke PROPERTIES
  PASS_REGULAR_EXPRESSION "Hello, World!"
)
